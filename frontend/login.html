sendCode.onclick = async () => { const phone = normalizePhone(mobile.value); if (!/^09\d{9}$/.test(phone)) { showMessage( "شماره موبایل اشتباه است", "error" ); return; } currentPhone = phone; mobile.value = phone; sendCode.disabled = true; showMessage( "در حال آماده‌سازی سرویس...", "" ); try { // اول Render را بیدار می‌کنیم await fetch( API + "/health", { method: "GET", cache: "no-store" } ); showMessage( "در حال ارسال کد...", "" ); const form = new FormData(); form.append( "mobile", phone ); const res = await fetch( API + "/send-code", { method: "POST", body: form, cache: "no-store" } ); const data = await res.json(); if (data.success) { showMessage( "کد ورود به همین شماره پیامک شد", "success" ); codeBox.classList.remove("hide"); code.focus(); startTimer(60); } else { sendCode.disabled = false; showMessage( data.message || "ارسال کد ناموفق بود", "error" ); } } catch (error) { console.error( "SEND CODE ERROR:", error ); sendCode.disabled = false; showMessage( "خطا در ارتباط با سرور", "error" ); } };
